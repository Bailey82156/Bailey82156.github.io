    <script>

        // Get the HTML elements
        const hero = document.getElementById("hero");

        const heroHeading = document.getElementById("heroHeading");

        const heroSubheading = document.getElementById("heroSubheading");

        const darkAppearance = document.getElementById("darkAppearance");

        const centreText = document.getElementById("centreText");

        const minimumHeight = document.getElementById("minimumHeight");

        const showSubheading = document.getElementById("showSubheading");

        const copyButton = document.getElementById("copyButton");

        const statusMessage = document.getElementById("statusMessage");

        const exportArea = document.getElementById("exportArea");

        const exportCode = document.getElementById("exportCode");


        // Store the four checkbox settings
        const settings = {
            dark: darkAppearance.checked,
            centre: centreText.checked,
            minHeight: minimumHeight.checked,
            showH2: showSubheading.checked
        };


        // Update the hero appearance without replacing its content
        function updateHero() {

            settings.dark = darkAppearance.checked;
            settings.centre = centreText.checked;
            settings.minHeight = minimumHeight.checked;
            settings.showH2 = showSubheading.checked;

            // Update light and dark appearance
            hero.classList.toggle("hero-dark", settings.dark);

            hero.classList.toggle("hero-light", !settings.dark);

            // Update text alignment
            hero.classList.toggle("hero-centre", settings.centre);

            // Update minimum height
            hero.classList.toggle("hero-min-height", settings.minHeight);

            // Show or hide the subheading
            heroSubheading.hidden = !settings.showH2;

            // Keep exported HTML updated
            updateExportCode();
        }


        // Escape special HTML characters in heading text
        function escapeHTML(text) {

            return text.replace(/[&<>"']/g, function(character) {

                const characters = {
                    "&": "&amp;",
                    "<": "&lt;",
                    ">": "&gt;",
                    '"': "&quot;",
                    "'": "&#39;"
                };

                return characters[character];

            });

        }


        // Generate a complete standalone HTML document
        function generateHTML() {

            const headingText = escapeHTML(heroHeading.textContent);

            const subheadingText = escapeHTML(heroSubheading.textContent);

            // Build the hero class list
            const heroClasses = [
                "hero",
                settings.dark ? "hero-dark" : "hero-light",
                settings.centre ? "hero-centre" : "",
                settings.minHeight ? "hero-min-height" : ""
            ].filter(Boolean).join(" ");

            // Include the H2 only when requested
            const subheadingHTML = settings.showH2
                ? `<h2>${subheadingText}</h2>`
                : "";

            // Return the complete HTML page
            return `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>My Hero Block</title>

    <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

    <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

    <style type="text/tailwindcss">

        .hero {
            @apply flex flex-col justify-center px-6 py-12 md:px-16;
        }

        .hero-content {
            @apply w-full max-w-5xl;
        }

        .hero-light {
            @apply bg-slate-50;
        }

        .hero-dark {
            @apply bg-slate-900;
        }

        .hero-centre {
            @apply text-center;
        }

        .hero-min-height {
            @apply min-h-[480px] md:min-h-[85vh];
        }

        .hero h1 {
            @apply mb-6 text-4xl font-bold tracking-tight text-slate-900 md:text-6xl break-words;
        }

        .hero h2 {
            @apply text-xl leading-relaxed text-slate-700 md:text-2xl break-words;
        }

        .hero-dark h1 {
            @apply text-slate-50;
        }

        .hero-dark h2 {
            @apply text-slate-300;
        }

        .hero-centre .hero-content {
            @apply mx-auto;
        }

        .hero h2[hidden] {
            display: none;
        }

    </style>

</head>

<body>

    <header class="${heroClasses}">
        <div class="hero-content">
            <h1>${headingText}</h1>
            ${subheadingHTML}
        </div>
    </header>

</body>
</html>`;

        }


        // Keep the fallback textarea up to date
        function updateExportCode() {

            exportCode.value = generateHTML();

        }


        // Copy the generated HTML
        async function copyHTML() {

            const generatedHTML = generateHTML();

            try {

                if (!navigator.clipboard || !window.isSecureContext) {
                    throw new Error("Clipboard unavailable");
                }

                await navigator.clipboard.writeText(generatedHTML);

                statusMessage.textContent =
                    "HTML copied—paste it into VS Code.";

                exportArea.hidden = true;

            } catch (error) {

                // Show the generated HTML for manual copying
                exportCode.value = generatedHTML;

                exportArea.hidden = false;

                statusMessage.textContent =
                    "Automatic copying is unavailable. Copy the HTML from the text area below.";

                exportCode.focus();

                exportCode.select();

            }

        }


        // Listen for checkbox changes
        darkAppearance.addEventListener("change", updateHero);

        centreText.addEventListener("change", updateHero);

        minimumHeight.addEventListener("change", updateHero);

        showSubheading.addEventListener("change", updateHero);


        // Update exported HTML when headings are edited
        heroHeading.addEventListener("input", updateExportCode);

        heroSubheading.addEventListener("input", updateExportCode);


        // Listen for the Copy HTML button
        copyButton.addEventListener("click", copyHTML);


        // Set the initial appearance
        updateHero();

    </script>

</body>
