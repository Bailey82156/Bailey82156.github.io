            h2: "slate-700"
        }

    };


    // CREATE THE COLOUR PALETTE

    function createPalette() {

        paletteGrid.innerHTML = "";

        // Create the shade heading row

        const emptyCell = document.createElement("div");
        paletteGrid.appendChild(emptyCell);

        colourShades.forEach(function(shade) {

            const heading = document.createElement("div");

            heading.className = "palette-header";
            heading.textContent = shade;

            paletteGrid.appendChild(heading);

        });


        // Create the colour family rows

        colourFamilies.forEach(function(family) {

            const familyLabel = document.createElement("div");

            familyLabel.className = "palette-family";

            familyLabel.textContent =
                family.charAt(0).toUpperCase() + family.slice(1);

            paletteGrid.appendChild(familyLabel);

            colourShades.forEach(function(shade) {

                const colourName = family + "-" + shade;

                const swatch = document.createElement("button");

                swatch.type = "button";
                swatch.className = "swatch";
                swatch.dataset.colour = colourName;

                swatch.setAttribute(
                    "aria-label",
                    "Select " + colourName
                );

                swatch.setAttribute("aria-pressed", "false");
                swatch.title = colourName;

                paletteGrid.appendChild(swatch);

            });

        });


        // Generate swatch CSS using Tailwind colour variables

        let swatchCSS = "";

        colourFamilies.forEach(function(family) {

            colourShades.forEach(function(shade) {

                const colourName = family + "-" + shade;

                swatchCSS += `
                    .swatch[data-colour="${colourName}"] {
                        background-color: var(--color-${colourName});
                    }
                `;

            });

        });

        swatchCSS += `
            .special-swatch[data-colour="black"] {
                background-color: var(--color-black);
            }

            .special-swatch[data-colour="white"] {
                background-color: var(--color-white);
            }
        `;

        paletteSwatchStyles.textContent = swatchCSS;

        updatePaletteSelection();

    }


    // UPDATE SELECTED COLOUR STYLES

    function updateHeroColours() {

        const backgroundRule = settings.gradientEnabled

            ? `
                background-color: transparent !important;
                background-image: linear-gradient(
                    ${settings.gradientDirection},
                    var(--color-${settings.colours.start}),
                    var(--color-${settings.colours.end})
                ) !important;
            `

            : `
                background-color: var(--color-${settings.colours.background}) !important;
                background-image: none !important;
            `;

        heroColorStyles.textContent = `

            .hero {
                ${backgroundRule}
            }

            .hero h1 {
                color: var(--color-${settings.colours.h1}) !important;
            }

            .hero h2 {
                color: var(--color-${settings.colours.h2}) !important;
            }

        `;

        updatePaletteSelection();
        updateExportCode();

    }


    // UPDATE PALETTE BUTTON STATES

    function updatePaletteSelection() {

        const allSwatches = document.querySelectorAll(
            ".swatch, .special-swatch"
        );

        allSwatches.forEach(function(swatch) {

            const selected =
                swatch.dataset.colour ===
                settings.colours[settings.activeTarget];

            swatch.setAttribute(
                "aria-pressed",
                selected ? "true" : "false"
            );

        });


        // Show only targets available in the current mode

        targetButtons.forEach(function(button) {

            const target = button.dataset.target;

            const visible =
                target === "h1" ||
                target === "h2" ||
                (
                    settings.gradientEnabled
                        ? target === "start" || target === "end"
                        : target === "background"
                );

            button.hidden = !visible;

            button.disabled =
                (target === "h2" && !settings.showH2) ||
                !visible;

            button.setAttribute(
                "aria-pressed",
                target === settings.activeTarget ? "true" : "false"
            );

        });


        const targetNames = {
            background: "Background",
            start: "Start colour",
            end: "End colour",
            h1: "H1",
            h2: "H2"
        };

        activeColour.textContent =
            "Active target: " +
            targetNames[settings.activeTarget].toUpperCase() +
            " | Selected colour: " +
            settings.colours[settings.activeTarget];

        gradientDirectionControl.hidden = !settings.gradientEnabled;

        gradientBackground.checked = settings.gradientEnabled;
        gradientDirection.value = settings.gradientDirection;

    }


    // SELECT COLOUR TARGET

    function selectTarget(target) {

        const validTargets = settings.gradientEnabled
            ? ["start", "end", "h1", "h2"]
            : ["background", "h1", "h2"];

        if (!validTargets.includes(target)) {
            return;
        }

        if (target === "h2" && !settings.showH2) {
            return;
        }

        settings.activeTarget = target;

        updatePaletteSelection();

    }


    // SELECT A COLOUR

    function selectColour(colourName) {

        const target = settings.activeTarget;

        settings.colours[target] = colourName;

        updateHeroColours();

    }


    // RESET COLOURS USING LIGHT OR DARK PRESET

    function updateColourPreset() {

        if (settings.dark) {

            settings.colours.background = "slate-900";
            settings.colours.h1 = "slate-50";
            settings.colours.h2 = "slate-300";

        } else {

            settings.colours.background = "slate-50";
            settings.colours.h1 = "slate-900";
            settings.colours.h2 = "slate-700";

        }

        // Gradient colours and direction are deliberately preserved.

        updateHeroColours();

    }


    // UPDATE HEADING LETTER SPACING

    function updateHeadingSpacing() {

        headingEffectsStyles.textContent = `

            .hero h1 {
                letter-spacing: ${settings.h1Spacing}em;
            }

            .hero h2 {
                letter-spacing: ${settings.h2Spacing}em;
            }

        `;

        h1SpacingValue.textContent =
            settings.h1Spacing.toFixed(3) + "em";

        h2SpacingValue.textContent =
            settings.h2Spacing.toFixed(3) + "em";

    }


    // UPDATE H2 EFFECT CONTROLS

    function updateH2Controls() {

        h2Serif.disabled = !settings.showH2;
        h2Spacing.disabled = !settings.showH2;

        if (!settings.showH2 && settings.activeTarget === "h2") {

            settings.activeTarget =
                settings.gradientEnabled ? "start" : "background";

        }

        updatePaletteSelection();

    }


    // SWITCH BETWEEN SOLID AND GRADIENT BACKGROUNDS

    function updateGradientMode() {

        const wasEnabled = settings.gradientEnabled;

        settings.gradientEnabled = gradientBackground.checked;

        // Initialise the gradient only the first time it is enabled.

        if (
            !wasEnabled &&
            settings.gradientEnabled &&
            !settings.gradientInitialised
        ) {

            settings.colours.start = settings.colours.background;
            settings.colours.end = "blue-500";

            settings.gradientInitialised = true;

        }

        // Keep the active target visible after switching modes.

        const visibleTargets = settings.gradientEnabled
            ? ["start", "end", "h1", "h2"]
            : ["background", "h1", "h2"];

        if (!visibleTargets.includes(settings.activeTarget)) {

            settings.activeTarget =
                settings.gradientEnabled ? "start" : "background";

        }

        updateHeroColours();

    }


    // UPDATE GRADIENT DIRECTION

    function updateGradientDirection() {

        settings.gradientDirection = gradientDirection.value;

        updateHeroColours();

    }


    // UPDATE HERO APPEARANCE

    function updateHero() {

        settings.dark = darkAppearance.checked;
        settings.centre = centreText.checked;
        settings.minHeight = minimumHeight.checked;
        settings.showH2 = showSubheading.checked;

        settings.h1Serif = h1Serif.checked;
        settings.h2Serif = h2Serif.checked;

        hero.classList.toggle("hero-dark", settings.dark);
        hero.classList.toggle("hero-light", !settings.dark);

        hero.classList.toggle("hero-centre", settings.centre);

        hero.classList.toggle(
            "hero-min-height",
            settings.minHeight
        );

        hero.classList.toggle(
            "hero-h1-serif",
            settings.h1Serif
        );

        hero.classList.toggle(
            "hero-h2-serif",
            settings.h2Serif
        );

        heroSubheading.hidden = !settings.showH2;

        updateH2Controls();
        updateExportCode();

    }


    // UPDATE H1 SPACING

    function updateH1Spacing() {

        settings.h1Spacing = Number(h1Spacing.value);

        updateHeadingSpacing();
        updateExportCode();

    }


    // UPDATE H2 SPACING

    function updateH2Spacing() {

        settings.h2Spacing = Number(h2Spacing.value);

        updateHeadingSpacing();
        updateExportCode();

    }


    // ESCAPE HTML CHARACTERS

    function escapeHTML(text) {

        return text.replace(/[&<>"']/g, function(character) {

            const characters = {
                "&": "&amp;",
                "<": "&lt;",
                ">": "&gt;",
                '"': "&quot;",
                "'": "&#39;"
            };

            return characters[character];

        });

    }


    // GENERATE COLOUR CSS FOR EXPORTED HTML

    function generateColourCSS() {

        const backgroundRule = settings.gradientEnabled

            ? `
                background-color: transparent !important;
                background-image: linear-gradient(
                    ${settings.gradientDirection},
                    var(--color-${settings.colours.start}),
                    var(--color-${settings.colours.end})
                ) !important;
            `

            : `
                background-color: var(--color-${settings.colours.background}) !important;
                background-image: none !important;
            `;

        return `

            .hero {
                ${backgroundRule}
            }

            .hero h1 {
                color: var(--color-${settings.colours.h1}) !important;
            }

            .hero h2 {
                color: var(--color-${settings.colours.h2}) !important;
            }

        `;

    }


    // GENERATE STANDALONE HTML

    function generateHTML() {

        const headingText = escapeHTML(heroHeading.textContent);
        const subheadingText = escapeHTML(heroSubheading.textContent);

        const heroClasses = [

            "hero",

            settings.dark ? "hero-dark" : "hero-light",

            settings.centre ? "hero-centre" : "",

            settings.minHeight ? "hero-min-height" : "",

            settings.h1Serif ? "hero-h1-serif" : "",

            settings.h2Serif ? "hero-h2-serif" : ""

        ].filter(Boolean).join(" ");


        const subheadingHTML = settings.showH2

            ? `<h2>${subheadingText}</h2>`

            : "";


        const exportedSpacingCSS = `

            .hero h1 {
                letter-spacing: ${settings.h1Spacing}em;
            }

            .hero h2 {
                letter-spacing: ${settings.h2Spacing}em;
            }

        `;


        const exportedColourCSS = generateColourCSS();


        return `<!DOCTYPE html>
<html lang="en">
<head>

    <meta charset="UTF-8">

    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>My Hero Block</title>

    <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

    <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

    <style type="text/tailwindcss">
        @source inline("bg-{slate,gray,zinc,neutral,stone,red,orange,amber,yellow,lime,green,emerald,teal,cyan,sky,blue,indigo,violet,purple,fuchsia,pink,rose}-{50,100,200,300,400,500,600,700,800,900,950} bg-black bg-white");
    </style>

    <style type="text/tailwindcss">

        .hero {
            @apply flex flex-col justify-center px-6 py-12 md:px-16;
        }

        .hero-content {
            @apply w-full max-w-5xl;
        }

        .hero-light {
            @apply bg-slate-50;
        }

        .hero-dark {
            @apply bg-slate-900;
        }

        .hero-centre {
            @apply text-center;
        }

        .hero-min-height {
            @apply min-h-[480px] md:min-h-[85vh];
        }

        .hero h1 {
            @apply mb-6 text-4xl font-bold tracking-tight text-slate-900 md:text-6xl break-words;
        }

        .hero h2 {
            @apply text-xl leading-relaxed text-slate-700 md:text-2xl break-words;
        }

        .hero-dark h1 {
            @apply text-slate-50;
        }

        .hero-dark h2 {
            @apply text-slate-300;
        }

            @apply mx-auto;
        }

        .hero h2[hidden] {
            display: none;
        }

        }

    </style>

            ${subheadingHTML}

        </div>

    </header>

</body>
</html>`;

    }


    // UPDATE FALLBACK EXPORT

    function updateExportCode() {

        exportCode.value = generateHTML();

    }


    // COPY HTML

    async function copyHTML() {

        const generatedHTML = generateHTML();

        try {

            if (!navigator.clipboard || !window.isSecureContext) {

                throw new Error("Clipboard unavailable");

            }

            await navigator.clipboard.writeText(generatedHTML);

            statusMessage.textContent =
                "HTML copied—paste it into VS Code.";

            exportArea.hidden = true;

        } catch (error) {

            exportCode.value = generatedHTML;

            exportArea.hidden = false;

            statusMessage.textContent =
                "Automatic copying is unavailable. Copy the HTML from the text area below.";

            exportCode.focus();
            exportCode.select();

        }

    }


    // EVENT LISTENERS

    darkAppearance.addEventListener("change", function() {

        settings.dark = darkAppearance.checked;

        updateColourPreset();
        updateHero();

    });

    centreText.addEventListener("change", updateHero);
    minimumHeight.addEventListener("change", updateHero);
    showSubheading.addEventListener("change", updateHero);

    gradientBackground.addEventListener(
        "change",
        updateGradientMode
    );

    gradientDirection.addEventListener(
        "change",
        updateGradientDirection
    );

    h1Serif.addEventListener("change", updateHero);
    h2Serif.addEventListener("change", updateHero);

    h1Spacing.addEventListener("input", updateH1Spacing);
    h2Spacing.addEventListener("input", updateH2Spacing);


    // PALETTE TARGET BUTTONS

    targetButtons.forEach(function(button) {

        button.addEventListener("click", function() {

            selectTarget(button.dataset.target);

        });

    });


    // PALETTE SWATCH BUTTONS

    paletteGrid.addEventListener("click", function(event) {

        const swatch = event.target.closest(".swatch");

        if (!swatch) {
            return;
        }

        selectColour(swatch.dataset.colour);

    });


    // BLACK AND WHITE BUTTONS

    specialSwatches.forEach(function(swatch) {

        swatch.addEventListener("click", function() {

            selectColour(swatch.dataset.colour);

        });

    });


    // UPDATE EXPORTED HTML WHEN HEADINGS CHANGE

    heroHeading.addEventListener("input", updateExportCode);
    heroSubheading.addEventListener("input", updateExportCode);


    // COPY BUTTON

    copyButton.addEventListener("click", copyHTML);


    // INITIALISE APPLICATION

    createPalette();
    updateHeadingSpacing();
    updateHero();
    updateHeroColours();

</script>
